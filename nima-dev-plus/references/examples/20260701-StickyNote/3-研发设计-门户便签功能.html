ServiceImpl 核心逻辑

@Service
public class StickyNoteServiceImpl extends ServiceImpl<StickyNoteMapper, StickyNote> implements StickyNoteService {

    @Override
    public List<StickyNote> listing(String userId) {
        QueryWrapper<StickyNote> qw = new QueryWrapper<>();
        qw.eq("user_id", userId);
        qw.orderByDesc("update_time");
        return list(qw);
    }

    @Override
    public void saveOrUpdateNote(StickyNote note) {
        if (note.getId() != null) {
            note.setUpdateTime(new Date());
            updateById(note);
        } else {
            note.setCreateTime(new Date());
            note.setUpdateTime(new Date());
            save(note);
        }
    }
}

3.5 Controller 设计

@Api(tags = "便签管理")
@RestController
@RequestMapping("/portal/stickyNote")
public class StickyNoteController {

    private final StickyNoteService stickyNoteService;

    public StickyNoteController(StickyNoteService stickyNoteService) {
        this.stickyNoteService = stickyNoteService;
    }

    @GetMapping("/listing")
    public Result<?> listing(HttpServletRequest request) {
        String userId = JwtUtil.getUserNameByToken(request);
        if (userId == null) return Result.error("未登录");
        return Result.OK(stickyNoteService.listing(userId));
    }

    @PostMapping("/add")
    public Result<?> add(HttpServletRequest request) {
        String userId = JwtUtil.getUserNameByToken(request);
        if (userId == null) return Result.error("未登录");
        StickyNote note = new StickyNote();
        note.setUserId(userId);
        note.setContent("");
        stickyNoteService.saveOrUpdateNote(note);
        return Result.OK(note);
    }

    @PutMapping("/edit")
    public Result<?> edit(@RequestBody StickyNote note, HttpServletRequest request) {
        String userId = JwtUtil.getUserNameByToken(request);
        if (userId == null) return Result.error("未登录");
        StickyNote existing = stickyNoteService.getById(note.getId());
        if (existing == null || !existing.getUserId().equals(userId)) {
            return Result.error("无权操作");
        }
        existing.setContent(note.getContent());
        stickyNoteService.saveOrUpdateNote(existing);
        return Result.OK("保存成功");
    }

    @DeleteMapping("/delete")
    public Result<?> delete(@RequestParam String id, HttpServletRequest request) {
        String userId = JwtUtil.getUserNameByToken(request);
        if (userId == null) return Result.error("未登录");
        StickyNote existing = stickyNoteService.getById(id);
        if (existing == null || !existing.getUserId().equals(userId)) {
            return Result.error("无权操作");
        }
        stickyNoteService.removeById(id);
        return Result.OK("删除成功");
    }
}

4.4 实时保存 Composable

// composables/useStickyNote.ts
import { ref, onMounted } from "vue";
import { ElMessage, ElMessageBox } from "element-plus";
import type { StickyNoteItem } from "../types";
import { getStickyNoteList, createStickyNote, updateStickyNote, deleteStickyNote } from "@/api/stickyNote";

export function useStickyNote() {
  const notes = ref<StickyNoteItem[]>([]);
  const loading = ref(false);

  // 防抖定时器映射：noteId -> timer
  const saveTimers = new Map<string, ReturnType<typeof setTimeout>>();

  /** 加载便签列表 */
  const loadNotes = async () => {
    loading.value = true;
    try {
      const res = await getStickyNoteList();
      if (res.success) {
        notes.value = (res.result?.records || res.result || []) as StickyNoteItem[];
      }
    } finally {
      loading.value = false;
    }
  };

  /** 创建便签 */
  const addNote = async () => {
    const res = await createStickyNote({ content: "" });
    if (res.success) {
      await loadNotes();
    }
  };

  /** 实时保存（防抖500ms） */
  const saveNote = (note: StickyNoteItem) => {
    const timer = saveTimers.get(note.id);
    if (timer) clearTimeout(timer);
    saveTimers.set(note.id, setTimeout(async () => {
      await updateStickyNote({ id: note.id, content: note.content });
      saveTimers.delete(note.id);
    }, 500));
  };

  /** 失焦保存（立即执行，取消防抖） */
  const flushSave = async (note: StickyNoteItem) => {
    const timer = saveTimers.get(note.id);
    if (timer) {
      clearTimeout(timer);
      saveTimers.delete(note.id);
    }
    await updateStickyNote({ id: note.id, content: note.content });
  };

  /** 删除便签 */
  const removeNote = async (id: string) => {
    await ElMessageBox.confirm("确定删除这条便签吗？", "删除确认", {
      type: "warning"
    });
    await deleteStickyNote(id);
    notes.value = notes.value.filter(n => n.id !== id);
    ElMessage.success("已删除");
  };

  onMounted(loadNotes);

  return { notes, loading, addNote, saveNote, flushSave, removeNote };
}
